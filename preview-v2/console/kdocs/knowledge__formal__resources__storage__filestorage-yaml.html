version: 1
resources:
  filestorage-volume:
    code: "st-fs-volume"
    service: storage/filestorage
    requires: []
    create:
      endpoint: "POST /v1/volumes"
      body: {name: "regrfs{unique}", protocol: NFS, type_name: HDD}
    capture: {volume_id: "$.volume_id"}
    ready: {field: "$.state", until: [available, AVAILABLE, ACTIVE],
            timeout: 300, interval: 15}
    verify:
      - {name: read-access-rules,
         endpoint: "GET /v1/volumes/{volume_id}/access-rules",
         expect_status: [200]}
    delete: {endpoint: "DELETE /v1/volumes/{volume_id}", destructive: true}
    provenance: VALIDATED
    source: {lifecycle: filestorage-volume, steps: [create, wait-available, delete]}
    notes: "standalone NFS volume (no vpc) — cross-service.yaml 키와 동일 노드. capture는 FLAT $.volume_id (virtualserver block volume의 $.id와 다름 — validated-facts). protocol NFS + type_name HDD가 유일하게 라이브 검증된 body (CIFS는 Windows용, 문서). 이름 규칙: 소문자 시작, 소문자/숫자/_, 3-21자 (services/storage__filestorage.yaml). 삭제 후 404 도달까지 폴링(소스 wait-gone 300s). ske-cluster의 volume_id 선행자원 (cross-service ske-volume-is-filestorage)."

    capture: {snapshot_id: "$.snapshot.id"}
    verify:
      - {name: read, endpoint: "GET /v1/snapshots/{snapshot_id}",
         expect_status: [200, 403],

      - {name: list,
         endpoint: "GET /v1/snapshots?volume_id={filestorage-volume.volume_id}",
         expect_status: [200],
         note: "volume_id 쿼리 필수 (docs + bare list 400 Field required, runs 27401527554/27417986669)"}
    delete: {endpoint: "DELETE /v1/snapshots/{snapshot_id}?volume_id={filestorage-volume.volume_id}",
             retry_on_status: [500, 409], retries: 4, retry_interval: 15,
             destructive: true}
    # delete도 volume_id 쿼리 필수 (docs + run 27421363609 400 Field required)
    provenance: VALIDATED  # gen-wave2-fs 전 단계 2xx — run 27424991237 (create/read/list/delete, capture $.snapshot.id 증명)

             steps: [create-snapshot, delete-snapshot]}
    notes: "capture envelope $.snapshot.id 라이브 증명 (run 27424991237 gen-wave2-fs). 볼륨당 최대 800 스냅샷(스케줄 생성분 128). 삭제는 소스에서 409/500 재시도 4×15s. restore는 PUT /v1/snapshots/{snapshot_id}/restore — 소유자 확인(2026-06-12): 스냅샷 생성은 볼륨만 있으면 즉시 가능, 복구는 스냅샷 생성 후에만 가능 (C4 verify 후보)."

    capture: {snapshot_schedule_id: "$.snapshot_schedule.id"}
    delete: {endpoint: "DELETE /v1/snapshot-schedules/{snapshot_schedule_id}",
             destructive: true}
    provenance: VALIDATED
    source: {lifecycle: filestorage-replication-schedule,
             steps: [create-snapshot-schedule, delete-snapshot-schedule]}
    notes: "KNOWN-HARD capture: SnapshotScheduleCreateResponse는 문서상 id를 안 돌려준다 (services quirk snapshot-schedule-no-id) — 소스도 capture_soft였고 미충전 placeholder로 set/delete를 호출했다. 실사용 시 list에서 id 회수 필요 (R3 과제: 합성기 list-recover 패턴). body는 api_docs SnapshotScheduleCreateRequest, UNPROVEN. live-validated run 27583285457 (2026-06-15)"

        replication_type: {type: enum, values: [replication, backup],
                 default: replication, vary: true,
                 note: "소유자 지식(2026-06-12): replication=타리전 DR 복제, backup=디스크백업(다른 FS 볼륨을 대상으로 — 두 번째 볼륨 선행 생성 필요). backup 변형의 대상 볼륨 필드 매핑은 유저가이드 확인 후 배선 (그 전까지 backup 변형 합성 금지)"}
    capture: {replication_id: "$.replication_id"}
    delete: {endpoint: "DELETE /v1/replications/{replication_id}", destructive: true}
    provenance: docs
    source: {lifecycle: filestorage-replication-schedule,

    notes: "body는 api_docs ReplicationCreateRequest, capture $.replication_id는 capture_soft (envelope UNPROVEN) — docs 유지. 소유자 지식(2026-06-12): 복제는 볼륨 생성 후 + 대상은 다른 리전(kr-east1). 디스크백업(replication_type=backup 추정)은 두 번째 FS 볼륨이 대상 — services fs-disk-backup-needs-second-volume."
